IF "DB_Data".Stanowisko1.Commands.AbortProcess.State THEN
    "DB_Data".Stanowisko1.Production.State := 0;
    "DB_Data".Stanowisko1.Commands.AbortProcess.State := 0;
END_IF;

CASE "DB_Data".Stanowisko1.Production.State OF
        
    0:
        //Ustawienie przyciskow
        "DB_Data".Stanowisko1.Commands.AbortProcess.Enable := 0;
        "DB_Data".Stanowisko1.Commands.FInish.Enable := 0;
        "DB_Data".Stanowisko1.Commands.ReportFault.Enable := 0;

	//Wyzerowanie stanu przyciskow (na wypadek zaleglych klikniec)
   	"DB_Data".Stanowisko1.Commands.FInish.State := 0;
    	"DB_Data".Stanowisko1.Commands.ReportFault.State := 0;
        
        //Ustawienie awarii na 0
        "DB_Data".Stanowisko1.Commands.FaultReason.DamagedPart := 0;
        "DB_Data".Stanowisko1.Commands.FaultReason.MissingPart := 0;
        "DB_Data".Stanowisko1.Commands.FaultReason.Other := 0;
        "DB_Data".Stanowisko1.Commands.FaultReason.PalletFault := 0;
        "DB_Data".Stanowisko1.Commands.FaultReason.SensorFault := 0;
        "DB_Data".Stanowisko1.Commands.FaultReason.WrongInstruction := 0;
        
	//Reset czasu cyklu na wyswietlaczu
    	"DB_Data".Stanowisko1.[Production].[CycleTime] := '00:00';

    	//Reset numeru zlecenia
    	"DB_Data".Stanowisko1.[Production].[OrderNumber] := 0;

	//Zerowanie flagi ladowania
	[Data loaded] := 0;

    //Sprawdzenie czy jest zlecenie (numer int = 0 -> brak zlecenia)
    IF "DB_Data".Stanowisko1.[Order].[LastNumber] = 0 THEN
        //Brak zlecenia - Start zablokowany
        "DB_Data".Stanowisko1.Commands.Start.Enable := 0;
        "DB_Data".Stanowisko1.[Status].[NoOrder] := 1;

    ELSE
        //Jest zlecenie
        "DB_Data".Stanowisko1.[Status].[NoOrder] := 0;

        //Przejscie do montazu jezeli paletka jest aktywna i start klikniety
        IF "DB_Data".Stanowisko1.Sensors.PalletPresent THEN
            "DB_Data".Stanowisko1.Commands.Start.Enable := 1;
            IF "DB_Data".Stanowisko1.Commands.Start.State THEN
                "DB_Data".Stanowisko1.Production.State := 1;
                "DB_Data".Stanowisko1.Commands.Start.State := 0;
            END_IF;
        ELSE
            "DB_Data".Stanowisko1.Commands.Start.Enable := 0;
        END_IF;
    END_IF;

    1:
        //Ustawienie przycisków
        "DB_Data".Stanowisko1.Commands.Start.Enable := 0;
        "DB_Data".Stanowisko1.Commands.AbortProcess.Enable := 1;
        "DB_Data".Stanowisko1.Commands.FInish.Enable := 1;
        "DB_Data".Stanowisko1.Commands.ReportFault.Enable := 1;

	//raz wykonywane na początku
	 IF NOT "DB_Data".Stanowisko1.[Status].[DataLoaded] THEN
        //Godzina rozpoczecia montazu - czas lokalny z zegara CPU (DTL)
        "RD_LOC_T"(OUT => "DB_Data".Stanowisko1.[Production].[StartTime]);

        //Zaczytanie czasu montazu dla tego stanowiska
        "DB_Data".Stanowisko1.[Production].[AssemblyTime] := "DB_Data".[Config].Stanowisko1.[AssemblyTime];

        //Dane zlecenia
        "DB_Data".Stanowisko1.[Production].[OrderNumber]  := "DB_Data".[Source].[OrderNumber];
        [sztuka] := [tez zaciągnąć]
        "DB_Data".Stanowisko1.[Production].[Instruction]  := "DB_Data".[Source].[Instruction];

        "DB_Data".Stanowisko1.[Status].[DataLoaded] := 1;
	END_IF;
	
	//--- Odliczanie czasu montazu w dol (co 1 s) ---
    //Generator impulsu 1 s (timer sam sie resetuje)
    #Tick(IN := NOT #Tick.Q, PT := T#1S);

    IF #Tick.Q THEN
        IF "DB_Data".Stanowisko1.[Production].[RemainingSec] > 0 THEN
            "DB_Data".Stanowisko1.[Production].[RemainingSec] := "DB_Data".Stanowisko1.[Production].[RemainingSec] - 1;
        END_IF;
    END_IF;

    //Generator mrugania 0,5 s -> przelacza flage Blink
    #BlinkTimer(IN := NOT #BlinkTimer.Q, PT := T#500MS);
    IF #BlinkTimer.Q THEN
        "DB_Data".Stanowisko1.[Status].[Blink] := NOT "DB_Data".Stanowisko1.[Status].[Blink];
    END_IF;

    IF "DB_Data".Stanowisko1.[Production].[RemainingSec] > 0 THEN
        //Normalne wyswietlanie MM:SS z zerem wiodacym
        #min := "DB_Data".Stanowisko1.[Production].[RemainingSec] / 60;
        #sec := "DB_Data".Stanowisko1.[Production].[RemainingSec] MOD 60;

        #minStr := INT_TO_STRING(#min);
        #secStr := INT_TO_STRING(#sec);
        IF #min < 10 THEN #minStr := CONCAT(IN1 := '0', IN2 := #minStr); END_IF;
        IF #sec < 10 THEN #secStr := CONCAT(IN1 := '0', IN2 := #secStr); END_IF;

        "DB_Data".Stanowisko1.[Production].[RemainingTime] := CONCAT(IN1 := #minStr, IN2 := CONCAT(IN1 := ':', IN2 := #secStr));
    ELSE
        //Czas minal - mruganie 00:00
        IF "DB_Data".Stanowisko1.[Status].[Blink] THEN
            "DB_Data".Stanowisko1.[Production].[RemainingTime] := '00:00';
        ELSE
            "DB_Data".Stanowisko1.[Production].[RemainingTime] := '';
        END_IF;
    END_IF;
        
        //Przejscie do zakończenia
        IF "DB_Data".Stanowisko1.Commands.FInish.State THEN
            "DB_Data".Stanowisko1.Production.State := 2;
            "DB_Data".Stanowisko1.Commands.FInish.State := 0;
        END_IF;
END_CASE;
